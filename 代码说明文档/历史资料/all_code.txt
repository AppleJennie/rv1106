static const char* state_to_string(system_state_e st)
{
    switch (st) {
        case STATE_BOOT:        return "BOOT";
        case STATE_READY:       return "READY";
        case STATE_RECORDING:   return "RECORDING";
        case STATE_UPLOADING:   return "UPLOADING";
        case STATE_UPLOAD_FAIL: return "UPLOAD_FAIL";
        case STATE_ERROR:       return "ERROR";
        default:                return "UNKNOWN";
    }
}

static uint64_t get_time_us(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts);
    return (uint64_t)ts.tv_sec * 1000000ULL + ts.tv_nsec / 1000ULL;
}

static void mkdir_if_not_exist(const char *path)
{
    struct stat st;
    if (stat(path, &st) == 0) return;
    mkdir(path, 0777);
}

static void ensure_storage_dirs(void)
{
    char photo_dir[MAX_PATH_LEN];
    char code_dir[MAX_PATH_LEN];

    mkdir_if_not_exist("/mnt/sd");
    mkdir_if_not_exist(SDCARD_BASE_PATH);

    snprintf(photo_dir, sizeof(photo_dir), "%s/photo", SDCARD_BASE_PATH);
    snprintf(code_dir, sizeof(code_dir), "%s/code", SDCARD_BASE_PATH);

    mkdir_if_not_exist(photo_dir);
    mkdir_if_not_exist(code_dir);
}

static void timestamp_to_second_name(uint64_t ts_us, char *buf, size_t len)
{
    time_t sec = (time_t)(ts_us / 1000000ULL);
    struct tm tm_now;
    localtime_r(&sec, &tm_now);
    strftime(buf, len, "%Y-%m-%d_%H-%M-%S", &tm_now);
}

static void clear_batch(second_batch_t *batch)
{
    if (!batch) return;

    for (int i = 0; i < batch->count; i++) {
        safe_free_ptr((void**)&batch->frames[i].img_data);
        batch->frames[i].img_size = 0;
    }
    batch->count = 0;
    memset(batch->second_name, 0, sizeof(batch->second_name));
}

static bool flush_batch_to_sdcard(second_batch_t *batch)
{
    if (!batch || batch->count <= 0) return true;

    char photo_dir[MAX_PATH_LEN];
    char csv_path[MAX_PATH_LEN];

    snprintf(photo_dir, sizeof(photo_dir), "%s/photo/%s", SDCARD_BASE_PATH, batch->second_name);
    mkdir_if_not_exist(photo_dir);

    snprintf(csv_path, sizeof(csv_path), "%s/code/%s.csv", SDCARD_BASE_PATH, batch->second_name);

    FILE *csv_fp = fopen(csv_path, "w");
    if (!csv_fp) {
        log_error("打开 CSV 失败: %s", csv_path);
        return false;
    }

    fprintf(csv_fp, "index,timestamp_us,enc1,enc2,enc3\n");

    for (int i = 0; i < batch->count; i++) {
        frame_data_t *f = &batch->frames[i];

        char img_path[MAX_PATH_LEN];
        snprintf(img_path, sizeof(img_path), "%s/%d.jpg", photo_dir, i + 1);

        FILE *img_fp = fopen(img_path, "wb");
        if (!img_fp) {
            log_error("打开图片失败: %s", img_path);
            fclose(csv_fp);
            return false;
        }

        fwrite(f->img_data, 1, f->img_size, img_fp);
        fclose(img_fp);

        fprintf(csv_fp, "%d,%llu,%.3f,%.3f,%.3f\n",
                i + 1,
                (unsigned long long)f->timestamp_us,
                f->enc_angle[0],
                f->enc_angle[1],
                f->enc_angle[2]);
    }

    fclose(csv_fp);

    log_info("已写入一秒数据: %s, 图片 %d 张, 编码器 %d 行",
             batch->second_name, batch->count, batch->count);

    clear_batch(batch);
    return true;
}

static void apply_led_by_state(system_state_e st)
{
    switch (st) {
        case STATE_BOOT:
            led_set_mode(LED_IDX_SYSTEM, LED_MODE_ON, LED_COLOR_GREEN, 0);
            led_set_mode(LED_IDX_STATUS, LED_MODE_OFF, LED_COLOR_OFF, 0);
            log_info("LED状态: 系统灯=绿色常亮, 状态灯=熄灭");
            break;

        case STATE_READY:
            led_set_mode(LED_IDX_SYSTEM, LED_MODE_ON, LED_COLOR_GREEN, 0);
            led_set_mode(LED_IDX_STATUS, LED_MODE_OFF, LED_COLOR_OFF, 0);
            log_info("LED状态: 系统灯=绿色常亮, 状态灯=熄灭");
            break;

        case STATE_RECORDING:
            led_set_mode(LED_IDX_SYSTEM, LED_MODE_BLINK, LED_COLOR_GREEN, 500);
            led_set_mode(LED_IDX_STATUS, LED_MODE_OFF, LED_COLOR_OFF, 0);
            log_info("LED状态: 系统灯=绿色闪烁, 状态灯=熄灭");
            break;

        case STATE_UPLOADING:
            led_set_mode(LED_IDX_SYSTEM, LED_MODE_ON, LED_COLOR_GREEN, 0);
            led_set_mode(LED_IDX_STATUS, LED_MODE_BLINK, LED_COLOR_BLUE, 300);
            log_info("LED状态: 系统灯=绿色常亮, 状态灯=蓝色闪烁");
            break;

        case STATE_UPLOAD_FAIL:
            led_set_mode(LED_IDX_SYSTEM, LED_MODE_ON, LED_COLOR_GREEN, 0);
            led_set_mode(LED_IDX_STATUS, LED_MODE_ON, LED_COLOR_RED, 0);
            log_info("LED状态: 系统灯=绿色常亮, 状态灯=红色常亮");
            break;

        case STATE_ERROR:
        default:
            led_set_mode(LED_IDX_SYSTEM, LED_MODE_ON, LED_COLOR_RED, 0);
            led_set_mode(LED_IDX_STATUS, LED_MODE_ON, LED_COLOR_RED, 0);
            log_info("LED状态: 系统灯=红色常亮, 状态灯=红色常亮");
            break;
    }
}

static void* capture_thread_func(void *arg)
{
    (void)arg;
    log_info("采集线程启动");
    clear_batch(&g_batch);

    while (g_capture_running) {
        frame_data_t frame;
        memset(&frame, 0, sizeof(frame));

        frame.timestamp_us = get_time_us();
        frame.frame_id = ++g_frame_id;

        if (!encoder_read_all(frame.enc_angle)) {
            log_warn("本帧编码器读取失败，继续采集");
        }

        if (!camera_grab_jpeg(&frame.img_data, &frame.img_size)) {
            log_warn("本帧图像抓取失败");
            usleep(CAPTURE_INTERVAL_US);
            continue;
        }

        char second_name[MAX_NAME_LEN];
        timestamp_to_second_name(frame.timestamp_us, second_name, sizeof(second_name));

        // 新的一秒到了，先把上一秒刷盘
        if (g_batch.count > 0 && strcmp(g_batch.second_name, second_name) != 0) {
            if (!flush_batch_to_sdcard(&g_batch)) {
                log_error("写入上一秒数据失败");
            }
        }

        // 新 batch 初始化
        if (g_batch.count == 0) {
            strncpy(g_batch.second_name, second_name, sizeof(g_batch.second_name) - 1);
        }

        // 存到 batch
        if (g_batch.count < FRAMES_PER_SECOND) {
            frame_data_t *dst = &g_batch.frames[g_batch.count];
            memset(dst, 0, sizeof(*dst));

            dst->timestamp_us = frame.timestamp_us;
            dst->frame_id = frame.frame_id;
            dst->enc_angle[0] = frame.enc_angle[0];
            dst->enc_angle[1] = frame.enc_angle[1];
            dst->enc_angle[2] = frame.enc_angle[2];
            dst->img_size = frame.img_size;
            dst->img_data = (uint8_t*)malloc(frame.img_size);

            if (!dst->img_data) {
                log_error("图片缓存分配失败");
                safe_free_ptr((void**)&frame.img_data);
                usleep(CAPTURE_INTERVAL_US);
                continue;
            }

            memcpy(dst->img_data, frame.img_data, frame.img_size);
            g_batch.count++;
        } else {
            log_warn("当前秒内帧数超过 30，丢弃额外帧");
        }

        safe_free_ptr((void**)&frame.img_data);

        if ((frame.frame_id % 30) == 0) {
            log_info("采集中: 已采集 %d 帧", frame.frame_id);
        }

        usleep(CAPTURE_INTERVAL_US);
    }

    // 停止时，把最后不足 30 帧也刷出去
    if (g_batch.count > 0) {
        if (!flush_batch_to_sdcard(&g_batch)) {
            log_error("停止采集时写入最后一批数据失败");
        }
    }

    log_info("采集线程退出，本次共采集 %d 帧", g_frame_id);
    return NULL;
}

static bool do_upload_mock(void)
{
    log_info("开始上传: 上传 %s 下的 code 和 photo 数据", SDCARD_BASE_PATH);
    return upload_all_sessions(SDCARD_BASE_PATH, &g_upload_cancel);
}

static void* upload_thread_func(void *arg)
{
    (void)arg;

    g_upload_done = 0;
    g_upload_result = 0;
    g_upload_cancel = 0;
    g_upload_running = true;

    bool ok = do_upload_mock();

    if (g_upload_cancel) {
        log_warn("上传被用户取消");
        g_upload_result = 0;
    } else if (ok) {
        log_info("上传成功");
        g_upload_result = 1;
    } else {
        log_error("上传失败");
        g_upload_result = 0;
    }

    g_upload_done = 1;
    g_upload_running = false;

    return NULL;
}

bool state_machine_init(void)
{
    log_info("状态机初始化开始");

    if (!gpio_init()) {
        log_error("GPIO 初始化失败");
        return false;
    }
    log_info("GPIO 初始化成功");

    if (!led_init()) {
        log_error("LED 初始化失败");
        return false;
    }
    log_info("LED 初始化成功");

    state_machine_set_state(STATE_BOOT);

    if (!camera_init()) {
        log_error("摄像头初始化失败");
        state_machine_set_state(STATE_ERROR);
        return false;
    }
    log_info("摄像头初始化成功");

    if (!encoder_init()) {
        log_error("编码器初始化失败");
        state_machine_set_state(STATE_ERROR);
        return false;
    }
    log_info("编码器初始化成功");

    if (!upload_init()) {
    log_error("上传模块初始化失败");
    state_machine_set_state(STATE_ERROR);
    return false;
}
log_info("上传模块初始化成功");

    ensure_storage_dirs();

    state_machine_set_state(STATE_READY);
    log_info("系统进入待机状态，等待按键触发");
    return true;
}

void state_machine_run(void)
{
    bool last_record_level = false;
    bool last_upload_level = false;

    uint64_t upload_debounce_ts = 0;

    while (g_run_flag) {
        led_tick();

        bool record_high = gpio_record_is_pressed();
        bool upload_high = gpio_upload_is_pressed();
        uint64_t now = get_time_us();

        // ========= 录制逻辑：高电平开始，低电平/悬空停止 =========
        if (record_high && !last_record_level) {
            if (g_system_state == STATE_READY || g_system_state == STATE_UPLOAD_FAIL) {
                log_info("录制控制输入变高: 开始采集");
                g_frame_id = 0;
                clear_batch(&g_batch);
                g_capture_running = true;

                if (pthread_create(&g_capture_thread, NULL, capture_thread_func, NULL) != 0) {
                    log_error("创建采集线程失败");
                    g_capture_running = false;
                    state_machine_set_state(STATE_ERROR);
                } else {
                    state_machine_set_state(STATE_RECORDING);
                }
            }
        }

        if (!record_high && last_record_level) {
            if (g_system_state == STATE_RECORDING) {
                log_info("录制控制输入变低/断开: 停止采集");
                g_capture_running = false;
                pthread_join(g_capture_thread, NULL);
                state_machine_set_state(STATE_READY);
            }
        }

        // ========= 上传逻辑：上传输入高电平触发一次 =========
        if (upload_high && !last_upload_level) {
            if (now - upload_debounce_ts > 300000) {
                upload_debounce_ts = now;

                if (g_system_state == STATE_READY || g_system_state == STATE_UPLOAD_FAIL) {
                    log_info("上传控制输入变高: 开始上传");

                    g_upload_cancel = 0;
                    g_upload_done = 0;
                    g_upload_result = 0;

                    if (pthread_create(&g_upload_thread, NULL, upload_thread_func, NULL) != 0) {
                        log_error("创建上传线程失败");
                        state_machine_set_state(STATE_UPLOAD_FAIL);
                    } else {
                        state_machine_set_state(STATE_UPLOADING);
                    }
                }
                else if (g_system_state == STATE_UPLOADING) {
                    log_warn("上传控制输入变高: 请求取消上传");
                    g_upload_cancel = 1;
                    upload_request_cancel();
                }
                else {
                    log_warn("当前状态 %s 下，上传输入无效", state_to_string(g_system_state));
                }
            }
        }

        /* 检查上传线程是否结束 */
        if (g_system_state == STATE_UPLOADING && g_upload_done) {
            pthread_join(g_upload_thread, NULL);
            g_upload_done = 0;

            if (g_upload_cancel) {
                log_warn("上传已取消，返回待机状态");
                state_machine_set_state(STATE_READY);
            } else if (g_upload_result) {
                log_info("上传完成，返回待机状态");
                state_machine_set_state(STATE_READY);
            } else {
                log_error("上传失败，进入失败状态");
                state_machine_set_state(STATE_UPLOAD_FAIL);
            }
        }

        last_record_level = record_high;
        last_upload_level = upload_high;

        usleep(10000);
    }
}

void state_machine_deinit(void)
{
    if (g_capture_running) {
        g_capture_running = false;
        pthread_join(g_capture_thread, NULL);
    }

    if (g_upload_running) {
        log_warn("程序退出: 正在取消上传线程");
        g_upload_cancel = 1;
        upload_request_cancel();
        pthread_join(g_upload_thread, NULL);
    }

    clear_batch(&g_batch);

    camera_deinit();
    encoder_deinit();
    led_deinit();
    gpio_deinit();
    upload_deinit();
    
    log_info("状态机已退出");
}

static int tca_select_channel(int ch)
{
    if (i2c_fd < 0) return -1;
    if (ioctl(i2c_fd, I2C_SLAVE, TCA9548A_ADDR) < 0) return -1;
    if (write(i2c_fd, &ch, 1) != 1) return -1;

// ==================== 读取 MT6701 角度 ====================
static float mt6701_read_angle(void)
{
    unsigned char buf[2];
    if (ioctl(i2c_fd, I2C_SLAVE, MT6701_ADDR) < 0) return -1.0f;
    
    buf[0] = 0x03;
    if (write(i2c_fd, buf, 1) != 1 || read(i2c_fd, buf, 1) != 1) return -1.0f;
    unsigned char bh = buf[0];
    
    buf[0] = 0x04;
    if (write(i2c_fd, buf, 1) != 1 || read(i2c_fd, buf, 1) != 1) return -1.0f;
    unsigned char bl = buf[0];
    
    unsigned short raw = ((unsigned short)bh << 6) | (bl & 0x3F);
    return (float)raw * 360.0f / 16384.0f;
}

bool encoder_init(void)

    struct stat st;
    while (stat(I2C_DEVICE, &st) != 0) {
        log_warn("等待 I2C 设备就绪...");
        sleep(1);
    }

    if (i2c_init_bus() != 0) {
        return false;
    }

    log_info("编码器预热中...");
    for (int i = 0; i < 10; i++) {
        for (int j = 0; j < ENC_COUNT; j++) {
            tca_select_channel(tca_ch_map[j]);
            mt6701_read_angle();
            usleep(500);
        }
    }

    for (int i = 0; i < ENC_COUNT; i++) {
        tca_select_channel(tca_ch_map[i]);
        float ang = mt6701_read_angle();
        if (ang >= 0) {
            log_info("%s：正常 %.1f°", enc_name[i], ang);
        } else {
            log_warn("%s：读取失败", enc_name[i]);
        }
    }

    log_info("编码器初始化完成");
    return true;
}

#include "sys_logger.h"

#define SPI_DEVICE    "/dev/spidev0.0"
#define SPI_SPEED_HZ  8000000
#define BIT_0         0xC0
#define BIT_1         0xFC
#define BYTES_PER_LED 24

typedef struct {
    led_mode_e mode;
    led_color_e color;
    int blink_period_ms;
    bool blink_on;
    uint64_t last_toggle_ms;
} led_ctrl_t;

static int spi_fd = -1;
static led_ctrl_t g_leds[LED_COUNT];
static pthread_mutex_t g_led_mutex = PTHREAD_MUTEX_INITIALIZER;

static uint64_t get_ms(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (uint64_t)ts.tv_sec * 1000ULL + ts.tv_nsec / 1000000ULL;
}

static void byte_to_spi(uint8_t data, uint8_t *out_buf)
{
    for (int i = 0; i < 8; i++) {
        out_buf[i] = (data & 0x80) ? BIT_1 : BIT_0;
        data <<= 1;
    }
}

static void color_to_grb(led_color_e color, uint8_t *g, uint8_t *r, uint8_t *b)
{
    *g = 0; *r = 0; *b = 0;
    switch (color) {
        case LED_COLOR_RED:    *r = 255; break;
        case LED_COLOR_GREEN:  *g = 255; break;
        case LED_COLOR_BLUE:   *b = 255; break;
        case LED_COLOR_YELLOW: *g = 255; *r = 255; break;
        default: break;
    }
}

static led_color_e get_effective_color(int i)
{
    if (g_leds[i].mode == LED_MODE_OFF) {
        return LED_COLOR_OFF;
    }
    if (g_leds[i].mode == LED_MODE_ON) {
        return g_leds[i].color;
    }
    if (g_leds[i].mode == LED_MODE_BLINK) {
        return g_leds[i].blink_on ? g_leds[i].color : LED_COLOR_OFF;
    }
    return LED_COLOR_OFF;
}

static void ws2812_send_current(void)
{
    if (spi_fd < 0) return;

    uint8_t spi_buf[LED_COUNT * BYTES_PER_LED];
    memset(spi_buf, 0, sizeof(spi_buf));

    for (int i = 0; i < LED_COUNT; i++) {
        uint8_t g, r, b;
        color_to_grb(get_effective_color(i), &g, &r, &b);

        uint8_t *p = &spi_buf[i * BYTES_PER_LED];
        byte_to_spi(g, p + 0);
        byte_to_spi(r, p + 8);
        byte_to_spi(b, p + 16);
    }

    struct spi_ioc_transfer xfer = {
        .tx_buf = (unsigned long)spi_buf,
        .len = sizeof(spi_buf),
        .speed_hz = SPI_SPEED_HZ,
        .bits_per_word = 8,
    };

    ioctl(spi_fd, SPI_IOC_MESSAGE(1), &xfer);
    usleep(100);
}

bool led_init(void)
{
    spi_fd = open(SPI_DEVICE, O_RDWR);
    if (spi_fd < 0) {
        log_error("打开 SPI 失败: %s", SPI_DEVICE);
        return false;
    }

    uint8_t mode = SPI_MODE_0;
    uint8_t bits = 8;
    uint32_t speed = SPI_SPEED_HZ;

    ioctl(spi_fd, SPI_IOC_WR_MODE, &mode);
    ioctl(spi_fd, SPI_IOC_WR_BITS_PER_WORD, &bits);
    ioctl(spi_fd, SPI_IOC_WR_MAX_SPEED_HZ, &speed);

    memset(g_leds, 0, sizeof(g_leds));
    ws2812_send_current();

    log_info("LED 初始化完成");
    return true;
}

void led_set_mode(int led_idx, led_mode_e mode, led_color_e color, int blink_period_ms)
{
    if (led_idx < 0 || led_idx >= LED_COUNT) return;

    pthread_mutex_lock(&g_led_mutex);

    g_leds[led_idx].mode = mode;
    g_leds[led_idx].color = color;
    g_leds[led_idx].blink_period_ms = blink_period_ms;
    g_leds[led_idx].blink_on = true;
    g_leds[led_idx].last_toggle_ms = get_ms();

    ws2812_send_current();

    pthread_mutex_unlock(&g_led_mutex);
}

void led_tick(void)
{
    pthread_mutex_lock(&g_led_mutex);

    uint64_t now = get_ms();
    bool changed = false;

    for (int i = 0; i < LED_COUNT; i++) {
        if (g_leds[i].mode == LED_MODE_BLINK && g_leds[i].blink_period_ms > 0) {
            int half_period = g_leds[i].blink_period_ms / 2;
            if ((int)(now - g_leds[i].last_toggle_ms) >= half_period) {
                g_leds[i].blink_on = !g_leds[i].blink_on;
                g_leds[i].last_toggle_ms = now;
                changed = true;
            }
        }
    }

    if (changed) {
        ws2812_send_current();
    }

    pthread_mutex_unlock(&g_led_mutex);
}

void led_deinit(void)
{
    pthread_mutex_lock(&g_led_mutex);

    for (int i = 0; i < LED_COUNT; i++) {
        g_leds[i].mode = LED_MODE_OFF;
        g_leds[i].color = LED_COLOR_OFF;
    }
    ws2812_send_current();

    pthread_mutex_unlock(&g_led_mutex);

    if (spi_fd >= 0) {
        close(spi_fd);
        spi_fd = -1;
    }

    log_info("LED 已关闭");
}

#include "sys_logger.h"

#define UPLOAD_SERVER_IP "192.168.4.17"
#define UPLOAD_SERVER_PORT 9000
#define RECV_BUF_SIZE      1024
#define SEND_BUF_SIZE      4096

static int g_sock = -1;
static volatile int *g_cancel_flag = NULL;

static bool path_exists(const char *path)
{
    struct stat st;
    return stat(path, &st) == 0;
}

static bool write_mark_file(const char *path)
{
    FILE *fp = fopen(path, "w");
    if (!fp) return false;
    fprintf(fp, "done\n");
    fclose(fp);
    return true;
}

static bool connect_server(void)
{
    struct sockaddr_in addr;
    g_sock = socket(AF_INET, SOCK_STREAM, 0);
    if (g_sock < 0) {
        log_error("创建上传 socket 失败");
        return false;
    }

    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_port = htons(UPLOAD_SERVER_PORT);
    if (inet_pton(AF_INET, UPLOAD_SERVER_IP, &addr.sin_addr) <= 0) {
        log_error("上传服务器 IP 无效: %s", UPLOAD_SERVER_IP);
        close(g_sock);
        g_sock = -1;
        return false;
    }

    if (connect(g_sock, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
        log_error("连接上传服务器失败: %s:%d", UPLOAD_SERVER_IP, UPLOAD_SERVER_PORT);
        close(g_sock);
        g_sock = -1;
        return false;
    }

    log_info("已连接上传服务器: %s:%d", UPLOAD_SERVER_IP, UPLOAD_SERVER_PORT);
    return true;
}

static void close_server(void)
{
    if (g_sock >= 0) {
        close(g_sock);
        g_sock = -1;
    }
}

static bool send_all(int fd, const void *buf, size_t len)
{
    const char *p = (const char*)buf;
    while (len > 0) {
        if (g_cancel_flag && *g_cancel_flag) {
            log_warn("send_all: 检测到取消请求");
            return false;
        }
        ssize_t n = send(fd, p, len, 0);
        if (n <= 0) return false;
        p += n;
        len -= n;
    }
    return true;
}

static bool recv_ok(int fd)
{
    if (g_cancel_flag && *g_cancel_flag) {
        return false;
    }

    char buf[RECV_BUF_SIZE];
    memset(buf, 0, sizeof(buf));
    ssize_t n = recv(fd, buf, sizeof(buf) - 1, 0);
    if (n <= 0) return false;
    return (strncmp(buf, "OK", 2) == 0);
}

static bool send_file(const char *full_path, const char *relative_path)
{
    if (g_cancel_flag && *g_cancel_flag) {
        return false;
    }

    struct stat st;
    if (stat(full_path, &st) != 0) {
        log_error("获取文件信息失败: %s", full_path);
        return false;
    }

    FILE *fp = fopen(full_path, "rb");
    if (!fp) {
        log_error("打开文件失败: %s", full_path);
        return false;
    }

    char header[1024];
    int header_len = snprintf(header, sizeof(header), "FILE %s %lld\n",
                              relative_path, (long long)st.st_size);
    if (header_len <= 0 || header_len >= (int)sizeof(header)) {
        fclose(fp);
        return false;
    }

    if (!send_all(g_sock, header, (size_t)header_len)) {
        log_error("发送文件头失败: %s", relative_path);
        fclose(fp);
        return false;
    }

    char buf[SEND_BUF_SIZE];
    size_t nread;
    while ((nread = fread(buf, 1, sizeof(buf), fp)) > 0) {
        if (g_cancel_flag && *g_cancel_flag) {
            log_warn("send_file: 检测到取消请求");
            fclose(fp);
            return false;
        }
        if (!send_all(g_sock, buf, nread)) {
            log_error("发送文件内容失败: %s", relative_path);
            fclose(fp);
            return false;
        }
    }

    fclose(fp);

    if (!recv_ok(g_sock)) {
        log_error("服务端未确认文件: %s", relative_path);
        return false;
    }

    log_info("上传成功: %s", relative_path);
    return true;
}

static int cmp_str(const void *a, const void *b)
{
    const char * const *sa = (const char * const *)a;
    const char * const *sb = (const char * const *)b;
    return strcmp(*sa, *sb);
}

static bool upload_one_group(const char *base_path, const char *group_name)
{
    if (g_cancel_flag && *g_cancel_flag) {
        return false;
    }

    char csv_path[MAX_PATH_LEN];
    char csv_done[MAX_PATH_LEN];
    char photo_dir[MAX_PATH_LEN];
    char photo_done[MAX_PATH_LEN];

    snprintf(csv_path, sizeof(csv_path), "%s/code/%s.csv", base_path, group_name);
    snprintf(csv_done, sizeof(csv_done), "%s/code/%s.csv.done", base_path, group_name);
    snprintf(photo_dir, sizeof(photo_dir), "%s/photo/%s", base_path, group_name);
    snprintf(photo_done, sizeof(photo_done), "%s/photo/%s/.done", base_path, group_name);

    if (path_exists(csv_done)) {
        log_info("跳过已上传组: %s", group_name);
        return true;
    }

    if (!path_exists(csv_path)) {
        log_warn("组缺少 CSV，跳过: %s", group_name);
        return true;
    }

    // 1. 先传 csv
    const char *rel_csv = strstr(csv_path, "/right/");
    if (rel_csv) rel_csv += 1; else rel_csv = csv_path;

    log_info("开始上传组: %s", group_name);

    if (!send_file(csv_path, rel_csv)) {
        log_error("上传组 CSV 失败: %s", group_name);
        return false;
    }

    // 2. 再传 photo 目录下 jpg
    DIR *dir = opendir(photo_dir);
    if (!dir) {
        log_error("打开照片目录失败: %s", photo_dir);
        return false;
    }

    // 先收集文件名并排序，保证 1.jpg -> 30.jpg 顺序
    char *names[256];
    int count = 0;

    struct dirent *ent;
    while ((ent = readdir(dir)) != NULL) {
        if (g_cancel_flag && *g_cancel_flag) {
            closedir(dir);
            for (int j = 0; j < count; j++) free(names[j]);
            return false;
        }

        if (strcmp(ent->d_name, ".") == 0 || strcmp(ent->d_name, "..") == 0) continue;
        if (strcmp(ent->d_name, ".done") == 0) continue;

        const char *dot = strrchr(ent->d_name, '.');
        if (!dot || strcmp(dot, ".jpg") != 0) continue;

        if (count < 256) {
            names[count] = strdup(ent->d_name);
            if (names[count]) count++;
        }
    }
    closedir(dir);

    qsort(names, count, sizeof(char*), cmp_str);

    for (int i = 0; i < count; i++) {
        if (g_cancel_flag && *g_cancel_flag) {
            for (int j = 0; j < count; j++) free(names[j]);
            return false;
        }

        char img_path[MAX_PATH_LEN];
        snprintf(img_path, sizeof(img_path), "%s/%s", photo_dir, names[i]);

        const char *rel_img = strstr(img_path, "/right/");
        if (rel_img) rel_img += 1; else rel_img = img_path;

        if (!send_file(img_path, rel_img)) {
            for (int j = 0; j < count; j++) free(names[j]);
            log_error("上传组图片失败: %s/%s", group_name, names[i]);
            return false;
        }
    }

    for (int i = 0; i < count; i++) free(names[i]);

    // 3. 全部成功后打标
    if (!write_mark_file(csv_done)) {
        log_warn("CSV 标记写入失败: %s", csv_done);
    }

    if (!write_mark_file(photo_done)) {
        log_warn("照片目录标记写入失败: %s", photo_done);
    }

    log_info("组上传完成并已标记: %s", group_name);
    return true;
}

bool upload_all_sessions(const char *base_path, volatile int *cancel_flag)

    }

    // 收集所有 csv 组名
    char *groups[512];
    int group_count = 0;

    struct dirent *ent;
    while ((ent = readdir(dir)) != NULL) {
        if (g_cancel_flag && *g_cancel_flag) {
            closedir(dir);
            for (int j = 0; j < group_count; j++) free(groups[j]);
            g_cancel_flag = NULL;
            close_server();
            return false;
        }

        if (strcmp(ent->d_name, ".") == 0 || strcmp(ent->d_name, "..") == 0) continue;

        const char *dot = strrchr(ent->d_name, '.');
        if (!dot || strcmp(dot, ".csv") != 0) continue;

        size_t len = (size_t)(dot - ent->d_name);
        if (len == 0 || len >= MAX_NAME_LEN) continue;

        if (group_count < 512) {
            groups[group_count] = (char*)malloc(len + 1);
            if (!groups[group_count]) continue;
            memcpy(groups[group_count], ent->d_name, len);
            groups[group_count][len] = '\0';
            group_count++;
        }
    }
    closedir(dir);

    qsort(groups, group_count, sizeof(char*), cmp_str);

    bool ok = true;
    for (int i = 0; i < group_count; i++) {
        if (g_cancel_flag && *g_cancel_flag) {
            ok = false;
            break;
        }
        if (!upload_one_group(base_path, groups[i])) {
            ok = false;
            break;
        }
    }

    for (int i = 0; i < group_count; i++) {
        free(groups[i]);
    }

    g_cancel_flag = NULL;
    close_server();
    return ok;
}

void upload_deinit(void)
{
    close_server();
}

static void log_print(log_level_e level, const char *fmt, va_list args)
{
    if (level < g_log_level) return;

    struct timeval tv;
    struct tm tm;
    gettimeofday(&tv, NULL);
    localtime_r(&tv.tv_sec, &tm);

    fprintf(stdout, "[%04d-%02d-%02d %02d:%02d:%02d.%03ld] [%s] ",
            tm.tm_year + 1900, tm.tm_mon + 1, tm.tm_mday,
            tm.tm_hour, tm.tm_min, tm.tm_sec, tv.tv_usec / 1000,
            level_str[level]);

    vfprintf(stdout, fmt, args);
    fprintf(stdout, "\n");
    fflush(stdout);
}

// 递归创建目录
static void create_dir_recursive(const char *path)
{
    char tmp[MAX_PATH_LEN];
    snprintf(tmp, sizeof(tmp), "%s", path);
    for (char *p = tmp + 1; *p; p++) {
        if (*p == '/') {
            *p = '\0';
            mkdir(tmp, 0777);
            *p = '/';
        }
    }
    mkdir(tmp, 0777);
}

bool storage_init(void)
{
    strncpy(g_base_path, SDCARD_BASE_PATH, MAX_PATH_LEN);

    if (!storage_check_sdcard()) {
        log_error("SD 卡检查失败");
        return false;
    }

    // 创建基础目录
    char photo_dir[MAX_PATH_LEN], code_dir[MAX_PATH_LEN];
    snprintf(photo_dir, MAX_PATH_LEN, "%s/photo", g_base_path);
    snprintf(code_dir, MAX_PATH_LEN, "%s/code", g_base_path);
    create_dir_recursive(photo_dir);
    create_dir_recursive(code_dir);

    // 初始化环形队列
    if (!ring_buffer_init(&g_ring_buffer)) {
        log_error("环形队列初始化失败");
        return false;
    }

    // 启动存储线程
    g_storage_thread_running = 1;
    if (pthread_create(&g_storage_thread, NULL, storage_thread_func, NULL) != 0) {
        log_error("存储线程创建失败");
        g_storage_thread_running = 0;
        ring_buffer_destroy(&g_ring_buffer);
        return false;
    }

    log_info("存储系统初始化完成，基础路径: %s", g_base_path);
    return true;
}

bool storage_write_frame(const char *session_path, const frame_data_t *frame)
{
    if (!session_path || !frame || !frame->img_data || frame->img_size <= 0) {
        log_error("写入帧数据无效");
        return false;
    }

    // 写入 JPEG 图片
    char img_path[MAX_PATH_LEN];
    snprintf(img_path, MAX_PATH_LEN, "%s/photo/%s/%d.jpg", g_base_path, session_path, frame->frame_id);
    FILE *img_fp = fopen(img_path, "wb");
    if (!img_fp) {
        log_error("图片文件打开失败: %s", img_path);
        return false;
    }
    fwrite(frame->img_data, 1, frame->img_size, img_fp);
    fclose(img_fp);

    // 追加写入编码器数据到 CSV
    char csv_path[MAX_PATH_LEN];
    snprintf(csv_path, MAX_PATH_LEN, "%s/code/%s/encoder_data.csv", g_base_path, session_path);
    FILE *csv_fp = fopen(csv_path, "a+");
    if (csv_fp) {
        if (ftell(csv_fp) == 0) {
            fprintf(csv_fp, "帧序号,时间戳(us),编码器1(°),编码器2(°),编码器3(°)\n");
        }
        fprintf(csv_fp, "%d,%llu,%.2f,%.2f,%.2f\n",
                frame->frame_id, frame->timestamp_us,
                frame->enc_angle[0], frame->enc_angle[1], frame->enc_angle[2]);
        fclose(csv_fp);
    }

    log_debug("第 %d 帧写入完成", frame->frame_id);
    return true;
}

bool ring_buffer_init(ring_buffer_t *rb)
{
    if (!rb) return false;
    memset(rb, 0, sizeof(ring_buffer_t));
    pthread_mutex_init(&rb->mutex, NULL);
    pthread_cond_init(&rb->cond, NULL);
    return true;
}

bool ring_buffer_push(ring_buffer_t *rb, const frame_data_t *frame)
{
    if (!rb || !frame) return false;

    pthread_mutex_lock(&rb->mutex);
    // 队列满，覆盖最旧帧
    if (rb->count >= RING_BUFFER_SIZE) {
        log_warn("环形队列已满，覆盖旧帧");
        frame_data_t *old_frame = &rb->buffer[rb->tail];
        safe_free(old_frame->img_data);
        rb->tail = (rb->tail + 1) % RING_BUFFER_SIZE;
        rb->count--;
    }

    // 拷贝帧数据
    frame_data_t *new_frame = &rb->buffer[rb->head];
    memcpy(new_frame, frame, sizeof(frame_data_t));
    new_frame->img_data = (uint8_t *)malloc(frame->img_size);
    if (!new_frame->img_data) {
        log_error("帧数据内存分配失败");
        pthread_mutex_unlock(&rb->mutex);
        return false;
    }
    memcpy(new_frame->img_data, frame->img_data, frame->img_size);

    rb->head = (rb->head + 1) % RING_BUFFER_SIZE;
    rb->count++;
    pthread_cond_signal(&rb->cond);
    pthread_mutex_unlock(&rb->mutex);
    return true;
}

bool ring_buffer_pop(ring_buffer_t *rb, frame_data_t *frame)
{
    if (!rb || !frame) return false;

    pthread_mutex_lock(&rb->mutex);
    while (rb->count <= 0 && g_storage_thread_running) {
        pthread_cond_wait(&rb->cond, &rb->mutex);
    }
    if (!g_storage_thread_running) {
        pthread_mutex_unlock(&rb->mutex);
        return false;
    }

    frame_data_t *buf_frame = &rb->buffer[rb->tail];
    memcpy(frame, buf_frame, sizeof(frame_data_t));
    buf_frame->img_data = NULL;

    rb->tail = (rb->tail + 1) % RING_BUFFER_SIZE;
    rb->count--;
    pthread_mutex_unlock(&rb->mutex);
    return true;
}

void ring_buffer_destroy(ring_buffer_t *rb)
{
    if (!rb) return;
    pthread_mutex_lock(&rb->mutex);
    for (int i = 0; i < RING_BUFFER_SIZE; i++) {
        safe_free(rb->buffer[i].img_data);
    }
    memset(rb->buffer, 0, sizeof(rb->buffer));
    rb->head = rb->tail = rb->count = 0;
    pthread_mutex_unlock(&rb->mutex);

    pthread_mutex_destroy(&rb->mutex);
    pthread_cond_destroy(&rb->cond);
}

void* storage_thread_func(void *arg)
{
    log_info("存储线程已启动");
    frame_data_t frame;
    char current_session[MAX_NAME_LEN] = {0};

    while (g_storage_thread_running) {
        if (ring_buffer_pop(&g_ring_buffer, &frame)) {
            storage_write_frame(current_session, &frame);
            safe_free(frame.img_data);
        }
    }

    // 退出前清空队列
    pthread_mutex_lock(&g_ring_buffer.mutex);
    while (g_ring_buffer.count > 0) {
        frame_data_t *buf_frame = &g_ring_buffer.buffer[g_ring_buffer.tail];
        storage_write_frame(current_session, buf_frame);
        safe_free(buf_frame->img_data);
        g_ring_buffer.tail = (g_ring_buffer.tail + 1) % RING_BUFFER_SIZE;
        g_ring_buffer.count--;
    }
    pthread_mutex_unlock(&g_ring_buffer.mutex);

    log_info("存储线程已退出");
    return NULL;
}

void storage_deinit(void)
{
    g_storage_thread_running = 0;
    pthread_cond_signal(&g_ring_buffer.cond);
    pthread_join(g_storage_thread, NULL);
    ring_buffer_destroy(&g_ring_buffer);
    log_info("存储系统已关闭");
}
